cmake_minimum_required(VERSION 3.22.1)
project(study_voice LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(VOICE_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../voicevox)
if(NOT EXISTS ${VOICE_ROOT}/include/voicevox_core.h)
 message(FATAL_ERROR "VOICEVOX CORE 0.17.0 official header missing; see docs/VOICEVOX.md")
endif()
add_library(voicevox_core SHARED IMPORTED)
set_target_properties(voicevox_core PROPERTIES IMPORTED_NO_SONAME TRUE IMPORTED_LOCATION ${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libvoicevox_core.so)
add_library(study_voice SHARED voice.cpp)
target_include_directories(study_voice PRIVATE ${VOICE_ROOT}/include)
target_link_libraries(study_voice voicevox_core)
target_link_options(study_voice PRIVATE "-Wl,-z,max-page-size=16384")
